胜 利

一、为什么“背景能动，但没有按钮 / 点不动”——排查步骤

  1. 看层级：背景动画对象是否盖在 Button 之上？UI 应在最顶层（Canvas Overlay / z-index 最高）。
  2. 看射线拦截：Unity 取消勾选 Raycast Target；Godot mouse_filter = IGNORE；Canvas pointer-events:none。90%“点不动”是这个。
  3. 看有没有渲染：按钮 alpha/scale=0 或被动画停帧？打日志确认 activeInHierarchy / visible。
  4. 看异步流程：加载存档/音效的 await 抛异常 → 后续 ShowMainUI 没执行。用 try/catch/finally 或 Promise.race(timeout) 保证 UI 必渲染。
  5. 看场景切换：onClick 绑了吗？EventSystem / Control 节点存在吗？
  6. 看分辨率适配：按钮被挤出屏？设 Canvas Scaler + 居中锚点。
  7. 最小复现：空场景只放背景+按钮，能点 = 遮挡/时序问题；不能点 = EventSystem 层级问题。

二、通用原则（本页已实现）

  • ① 交互层 z-index:9999 置顶，背景 pointer-events:none（= RaycastTarget=false）。
  • ② 资源加载无论成败，finally 强制渲染“开始游戏”按钮，3 秒超时兜底。
  • ③ “开始”同步切屏到种族选择 → 地图选择，无过场动画依赖。

三、Unity C# 修复代码

HomeUIController.cs
using System.Collections;
using UnityEngine;
using UnityEngine.UI;

public class HomeUIController : MonoBehaviour
{
    public Canvas uiCanvas;
    public Image backgroundAnim;
    public GameObject racePanel, mapPanel, homePanel;

    IEnumerator Start()
    {
        // 关键1：关闭背景射线拦截 + UI 置顶
        backgroundAnim.raycastTarget = false;
        uiCanvas.overrideSorting = true;
        uiCanvas.sortingOrder = 32767;

        // 关键2：兜底渲染主按钮（即使加载失败）
        var btnStart = homePanel.GetComponentInChildren<Button>(true);
        btnStart.gameObject.SetActive(false);
        bool ok = false;
        yield return StartCoroutine(LoadResWithTimeout(3f, r => ok = r));
        btnStart.gameObject.SetActive(true);       // 无论成败都显示
        if (!ok) Debug.LogWarning("[HomeUI] 资源加载失败/超时，已用兜底 UI");

        btnStart.onClick.RemoveAllListeners();
        btnStart.onClick.AddListener(EnterRaceSelect);   // 关键3
    }

    IEnumerator LoadResWithTimeout(float t, System.Action<bool> done)
    {
        float timer = 0;
        var op = AudioBundle.LoadAsync();   // 你的加载
        while (!op.isDone && timer < t) { timer += Time.deltaTime; yield return null; }
        done(op.isDone);
    }

    public void EnterRaceSelect()
    {
        homePanel.SetActive(false);
        racePanel.SetActive(true);          // 人/夜/兽/鬼
        mapPanel.SetActive(false);
    }
    public void EnterMapSelect(string race) { racePanel.SetActive(false); mapPanel.SetActive(true); }
}

四、Godot 4 修复代码（GDScript）

home_ui.gd
extends Control

@onready var home = %HomePanel
@onready var race = %RacePanel
@onready var map  = %MapPanel
@onready var bg   = $BackgroundAnim
@onready var btn  = %BtnStart

func _ready() -> void:
    # 关键1：背景不接收鼠标 + UI 顶层
    if bg is TextureRect: bg.mouse_filter = Control.MOUSE_FILTER_IGNORE
    z_index = 100
    # 关键2：兜底渲染开始按钮（3 秒超时也不拦 UI）
    btn.visible = false
    var ok = await _load_with_timeout(3.0)
    btn.visible = true              # 无论成败都显示
    if not ok: push_warning("资源加载失败，使用兜底 UI")
    btn.pressed.connect(enter_race_select)   # 关键3

func _load_with_timeout(t: float) -> bool:
    var task = _load_save_async()
    var timer = 0.0
    while not task.is_completed() and timer < t:
        timer += get_process_delta_time()
        await get_tree().process_frame
    return task.is_completed()

func enter_race_select() -> void:
    home.visible = false; race.visible = true
func enter_map_select() -> void:
    race.visible = false; map.visible = true

// 关键1：交互层置顶，背景关闭射线拦截
bgLayer.style.pointerEvents = 'none';   // = raycastTarget=false
uiLayer.style.zIndex = 9999;

// 关键2：异步容错 —— 超时/失败也强制渲染开始按钮
async function boot() {
  const btn = document.getElementById('btn-start');
  try {
    await Promise.race([
      Promise.all([loadData(), loadAudio()]),
      new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), 3000)),
    ]);
  } catch (e) {
    document.getElementById('start-fallback-note').style.display = 'block';
  } finally {
    btn.style.display = 'inline-block';   // 兜底：一定显示
  }
}
boot();
